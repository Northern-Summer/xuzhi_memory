#!/bin/bash
# ============================================
# OpenClaw 升级保护系统
# 确保自定义配置经受住官方升级
# ============================================
# 工程改进铁律合规 — Ξ | 2026-03-31
# 自问：此操作是否让系统更安全/准确/优雅/高效？答案：YES
# ============================================

set -e

CUSTOM_DIR="$HOME/.xuzhi_custom"
OPENCLAW_DIR="$HOME/.openclaw"
WORKSPACE_DIR="$HOME/.openclaw/workspace"
BACKUP_DIR="$CUSTOM_DIR/backups"
LOG_FILE="$CUSTOM_DIR/upgrade.log"

# 颜色定义
RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
BLUE='\033[0;34m'
NC='\033[0m'

log() {
    echo "[$(date '+%Y-%m-%d %H:%M:%S')] $1" >> "$LOG_FILE"
    echo -e "$1"
}

# ============================================
# 升级前备份
# ============================================
pre_upgrade_backup() {
    log "${BLUE}=== 升级前备份 ===${NC}"
    
    local timestamp=$(date +%Y%m%d_%H%M%S)
    local backup_path="$BACKUP_DIR/pre_upgrade_$timestamp"
    
    mkdir -p "$backup_path"
    
    # 备份 openclaw.json
    if [[ -f "$OPENCLAW_DIR/openclaw.json" ]]; then
        cp "$OPENCLAW_DIR/openclaw.json" "$backup_path/"
        log "${GREEN}✓${NC} openclaw.json 已备份"
    fi
    
    # 备份 workspace 中的自定义文件
    for file in "$WORKSPACE_DIR"/*.md; do
        if [[ -f "$file" ]] && [[ ! -L "$file" ]]; then
            cp "$file" "$backup_path/"
            log "${GREEN}✓${NC} $(basename $file) 已备份"
        fi
    done
    
    # 备份 scripts（如果不是 symlink）
    if [[ -d "$WORKSPACE_DIR/scripts" ]] && [[ ! -L "$WORKSPACE_DIR/scripts" ]]; then
        cp -r "$WORKSPACE_DIR/scripts" "$backup_path/"
        log "${GREEN}✓${NC} scripts/ 已备份"
    fi
    
    # 备份 memory 索引
    if [[ -d "$OPENCLAW_DIR/memory" ]]; then
        cp -r "$OPENCLAW_DIR/memory" "$backup_path/"
        log "${GREEN}✓${NC} memory 索引已备份"
    fi
    
    # 记录当前版本
    openclaw --version > "$backup_path/version.txt" 2>&1 || echo "unknown" > "$backup_path/version.txt"
    
    log "${GREEN}备份完成: $backup_path${NC}"
    echo "$backup_path"
}

# ============================================
# 升级后恢复
# ============================================
post_upgrade_restore() {
    log "${BLUE}=== 升级后恢复 ===${NC}"
    
    # 1. 恢复 openclaw.json（合并策略）
    if [[ -f "$CUSTOM_DIR/configs/openclaw.json.backup" ]]; then
        # 检查是否有需要合并的新配置
        if [[ -f "$OPENCLAW_DIR/openclaw.json" ]]; then
            # 保留官方新配置，但检查关键字段
            log "${YELLOW}!${NC} openclaw.json 需要手动检查合并"
            log "  官方配置: $OPENCLAW_DIR/openclaw.json"
            log "  我们的备份: $CUSTOM_DIR/configs/openclaw.json.backup"
        fi
    fi
    
    # 2. 确保 symlink 存在
    restore_symlinks
    
    # 3. 恢复 skills
    restore_skills
    
    # 4. 恢复 scripts
    restore_scripts
    
    log "${GREEN}恢复完成${NC}"
}

# ============================================
# 恢复 Symlink
# ============================================
restore_symlinks() {
    log "${BLUE}恢复 symlink...${NC}"
    
    # 核心配置文件 symlink
    local symlinks=(
        "AGENTS.md:.xuzhi_memory/agents/xi/AGENTS.md"
        "MEMORY.md:.xuzhi_memory/agents/xi/MEMORY.md"
        "SOUL.md:.xuzhi_memory/agents/xi/SOUL.md"
        "IDENTITY.md:.xuzhi_memory/agents/xi/IDENTITY.md"
        "TOOLS.md:.xuzhi_memory/agents/xi/TOOLS.md"
        "USER.md:.xuzhi_memory/agents/xi/USER.md"
        "HEARTBEAT.md:.xuzhi_memory/agents/xi/HEARTBEAT.md"
        "MEMORY_HYGIENE_RULES.md:.xuzhi_memory/MEMORY_HYGIENE_RULES.md"
    )
    
    for entry in "${symlinks[@]}"; do
        local target="${entry%%:*}"
        local source="${entry#*:}"
        local full_path="$WORKSPACE_DIR/$target"
        
        # 如果不是 symlink，创建它
        if [[ ! -L "$full_path" ]]; then
            # 备份现有文件
            if [[ -f "$full_path" ]]; then
                mv "$full_path" "$full_path.upgrade_backup"
                log "  ${YELLOW}!${NC} $target 已备份为 $target.upgrade_backup"
            fi
            ln -sf "$HOME/$source" "$full_path"
            log "  ${GREEN}✓${NC} $target → $source"
        else
            log "  ${GREEN}✓${NC} $target (已存在)"
        fi
    done
}

# ============================================
# 恢复 Skills
# ============================================
restore_skills() {
    log "${BLUE}恢复 skills...${NC}"
    
    mkdir -p "$OPENCLAW_DIR/skills"
    
    for skill_dir in "$CUSTOM_DIR/skills"/*; do
        if [[ -d "$skill_dir" ]]; then
            local skill_name=$(basename "$skill_dir")
            local target="$OPENCLAW_DIR/skills/$skill_name"
            
            # 如果不是 symlink，创建它
            if [[ ! -L "$target" ]]; then
                rm -rf "$target" 2>/dev/null || true
                ln -sf "$skill_dir" "$target"
                log "  ${GREEN}✓${NC} skill: $skill_name"
            else
                log "  ${GREEN}✓${NC} skill: $skill_name (已存在)"
            fi
        fi
    done
}

# ============================================
# 恢复 Scripts
# ============================================
restore_scripts() {
    log "${BLUE}恢复 scripts...${NC}"
    
    # scripts 不用 symlink，因为 OpenClaw 不管理这个目录
    # 直接复制即可
    mkdir -p "$WORKSPACE_DIR/scripts"
    
    for script in "$CUSTOM_DIR/scripts"/*; do
        if [[ -f "$script" ]]; then
            local script_name=$(basename "$script")
            cp "$script" "$WORKSPACE_DIR/scripts/"
            log "  ${GREEN}✓${NC} script: $script_name"
        fi
    done
}

# ============================================
# 验证自定义内容完整性
# ============================================
verify_integrity() {
    log "${BLUE}=== 验证自定义内容完整性 ===${NC}"
    
    local errors=0
    
    # 检查 symlink
    local critical_symlinks=(
        "$WORKSPACE_DIR/AGENTS.md"
        "$WORKSPACE_DIR/MEMORY.md"
        "$WORKSPACE_DIR/SOUL.md"
    )
    
    for link in "${critical_symlinks[@]}"; do
        if [[ -L "$link" ]] && [[ -f "$link" ]]; then
            log "  ${GREEN}✓${NC} $(basename $link) symlink 正常"
        else
            log "  ${RED}✗${NC} $(basename $link) symlink 异常"
            ((errors++))
        fi
    done
    
    # 检查自定义目录
    if [[ -d "$CUSTOM_DIR" ]] && [[ -d "$CUSTOM_DIR/.git" ]]; then
        log "  ${GREEN}✓${NC} 自定义目录版本控制正常"
    else
        log "  ${RED}✗${NC} 自定义目录版本控制异常"
        ((errors++))
    fi
    
    # 检查 skills
    if [[ -L "$OPENCLAW_DIR/skills/fork-subagent" ]]; then
        log "  ${GREEN}✓${NC} fork-subagent skill 正常"
    else
        log "  ${RED}✗${NC} fork-subagent skill 异常"
        ((errors++))
    fi
    
    if [[ $errors -eq 0 ]]; then
        log "${GREEN}完整性验证通过${NC}"
        return 0
    else
        log "${RED}完整性验证失败: $errors 个错误${NC}"
        return 1
    fi
}

# ============================================
# 显示帮助
# ============================================
show_help() {
    cat << 'EOF'
OpenClaw 升级保护系统

Usage:
  ./upgrade-protection.sh <command>

Commands:
  backup      升级前备份
  restore     升级后恢复
  verify      验证完整性
  symlinks    仅恢复 symlink
  skills      仅恢复 skills
  scripts     仅恢复 scripts
  help        显示帮助

升级流程:
  1. npm update -g openclaw 之前
     → ./upgrade-protection.sh backup
     
  2. npm update -g openclaw
     
  3. 升级后
     → ./upgrade-protection.sh restore
     → ./upgrade-protection.sh verify

自定义内容存储位置:
  ~/.xuzhi_custom/
  ├── scripts/    # Bash 安全增强等
  ├── skills/     # Fork Subagent 等
  ├── configs/    # openclaw.json 备份
  ├── docs/       # Fusion 文档
  └── backups/    # 升级前备份
EOF
}

# ============================================
# 主入口
# ============================================
mkdir -p "$BACKUP_DIR"

case "${1:-help}" in
    backup)
        pre_upgrade_backup
        ;;
    restore)
        post_upgrade_restore
        ;;
    verify)
        verify_integrity
        ;;
    symlinks)
        restore_symlinks
        ;;
    skills)
        restore_skills
        ;;
    scripts)
        restore_scripts
        ;;
    help|--help|-h)
        show_help
        ;;
    *)
        echo "未知命令: $1"
        show_help
        exit 1
        ;;
esac
