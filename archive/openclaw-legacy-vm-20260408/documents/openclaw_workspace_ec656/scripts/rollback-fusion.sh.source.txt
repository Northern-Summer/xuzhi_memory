#!/bin/bash
# ============================================
# OpenClaw Fusion Rollback Script
# 安全回滚到融合前状态
# ============================================

set -e

RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
NC='\033[0m'

BACKUP_DIR="$HOME/.openclaw/workspace/.backup"
WORKSPACE="$HOME/.openclaw/workspace"

echo "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
echo "  OpenClaw Fusion Rollback"
echo "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"

# 显示帮助
show_help() {
    echo "Usage: $0 <phase|all>"
    echo ""
    echo "Phases:"
    echo "  0   - Rollback preparation (restore backups)"
    echo "  1   - Rollback Bash security enhancements"
    echo "  2   - Rollback Fork Subagent"
    echo "  3   - Rollback QueryEngine adapter"
    echo "  4   - Rollback Prompt Caching"
    echo "  5   - Rollback Coordinator mode"
    echo "  all - Complete rollback to pre-fusion state"
    echo ""
    echo "Example: $0 1"
}

# 检查参数
if [[ -z "$1" ]]; then
    show_help
    exit 1
fi

PHASE="$1"

# Phase 0 回滚：恢复备份
rollback_phase_0() {
    echo -e "${YELLOW}[Phase 0] Restoring backups...${NC}"
    
    # 恢复 agents 备份
    if [[ -d "$BACKUP_DIR/agents" ]]; then
        echo "  Restoring agents..."
        rm -rf "$WORKSPACE/agents" 2>/dev/null || true
        cp -r "$BACKUP_DIR/agents" "$WORKSPACE/"
        echo -e "  ${GREEN}✓${NC} agents restored"
    fi
    
    # 恢复配置备份
    if [[ -f "$BACKUP_DIR/openclaw.json" ]]; then
        echo "  Restoring openclaw.json..."
        cp "$BACKUP_DIR/openclaw.json" "$HOME/.openclaw/openclaw.json"
        echo -e "  ${GREEN}✓${NC} openclaw.json restored"
    fi
    
    echo -e "${GREEN}Phase 0 rollback complete${NC}"
}

# Phase 1 回滚：移除 Bash 安全增强
rollback_phase_1() {
    echo -e "${YELLOW}[Phase 1] Removing Bash security enhancements...${NC}"
    
    local script="$WORKSPACE/scripts/bash-security-enhanced.sh"
    if [[ -f "$script" ]]; then
        rm "$script"
        echo -e "  ${GREEN}✓${NC} bash-security-enhanced.sh removed"
    fi
    
    echo -e "${GREEN}Phase 1 rollback complete${NC}"
}

# Phase 2 回滚：移除 Fork Subagent
rollback_phase_2() {
    echo -e "${YELLOW}[Phase 2] Removing Fork Subagent skill...${NC}"
    
    local skill="$HOME/.openclaw/skills/fork-subagent"
    if [[ -d "$skill" ]]; then
        rm -rf "$skill"
        echo -e "  ${GREEN}✓${NC} fork-subagent skill removed"
    fi
    
    echo -e "${GREEN}Phase 2 rollback complete${NC}"
}

# Phase 3 回滚：移除 QueryEngine 适配层
rollback_phase_3() {
    echo -e "${YELLOW}[Phase 3] Removing QueryEngine adapter...${NC}"
    
    local lib="$WORKSPACE/lib/query-engine"
    if [[ -d "$lib" ]]; then
        rm -rf "$lib"
        echo -e "  ${GREEN}✓${NC} query-engine removed"
    fi
    
    echo -e "${GREEN}Phase 3 rollback complete${NC}"
}

# 完全回滚
rollback_all() {
    echo -e "${RED}⚠️  FULL ROLLBACK - This will restore to pre-fusion state${NC}"
    read -p "Are you sure? (yes/no): " confirm
    
    if [[ "$confirm" != "yes" ]]; then
        echo "Rollback cancelled"
        exit 0
    fi
    
    rollback_phase_3
    rollback_phase_2
    rollback_phase_1
    rollback_phase_0
    
    # 重启 Gateway
    echo ""
    echo -e "${YELLOW}Restarting Gateway...${NC}"
    openclaw gateway restart
    
    echo ""
    echo -e "${GREEN}━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━${NC}"
    echo -e "${GREEN}  Full rollback complete${NC}"
    echo -e "${GREEN}━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━${NC}"
}

# 执行回滚
case "$PHASE" in
    0) rollback_phase_0 ;;
    1) rollback_phase_1 ;;
    2) rollback_phase_2 ;;
    3) rollback_phase_3 ;;
    4|5) echo "Phase $PHASE not yet implemented" ;;
    all) rollback_all ;;
    -h|--help) show_help ;;
    *) echo -e "${RED}Unknown phase: $PHASE${NC}"; show_help; exit 1 ;;
esac

echo ""
echo "Run 'openclaw gateway restart' to apply changes"
