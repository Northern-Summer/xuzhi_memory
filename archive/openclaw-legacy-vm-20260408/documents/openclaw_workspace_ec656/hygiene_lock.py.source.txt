"""
hygiene_lock.py — 记忆卫生锁，防止竞态
防止 morning_memory_hygiene 和 boot_memory_hygiene 同时跑
防止 rotation_driver 和 hygiene 同时跑
"""

import fcntl
import os
import sys
from pathlib import Path

LOCK_FILE = Path.home() / ".xuzhi_memory" / ".hygiene.lock"


def acquire_lock():
    """尝试获取锁，返回是否成功"""
    try:
        LOCK_FILE.parent.mkdir(parents=True, exist_ok=True)
        LOCK_FILE.touch(exist_ok=True)
        fd = os.open(str(LOCK_FILE), os.O_RDWR)
        try:
            fcntl.flock(fd, fcntl.LOCK_EX | fcntl.LOCK_NB)
            LOCK_FILE.write_text(f"{os.getpid()}")
            return True
        except BlockingIOError:
            # 另一个进程持有锁
            return False
        finally:
            os.close(fd)
    except Exception:
        return False


def release_lock():
    """释放锁"""
    try:
        LOCK_FILE.unlink()
    except Exception:
        pass


if __name__ == "__main__":
    if "--release" in sys.argv:
        release_lock()
        print("🔓 Lock released")
    else:
        if acquire_lock():
            print("🔐 Lock acquired")
        else:
            print("⚠️ Lock held by another process, exiting")
            sys.exit(1)
