import json, urllib.request, urllib.error, time, uuid
from datetime import datetime

ACTIVITY_PATH = "/home/summer/.xuzhi_memory/expert_tracker/activity.json"
BASE_URL = "https://ilinkai.weixin.qq.com"
TOKEN = "ae89b339f0f0@im.bot:060000006c1cc52a6916370690cc632bf8a898"
TO_USER = "o9cq80z9eOrqJasg6hB1W-Cc4-Po@im.wechat"

def get_context_token():
    """Read current get_updates_buf from sync.json as context_token"""
    sync_path = f"/home/summer/.openclaw/openclaw-weixin/accounts/ae89b339f0f0-im-bot.sync.json"
    try:
        with open(sync_path) as f:
            d = json.load(f)
        return d.get("get_updates_buf", "")
    except:
        return ""

def build_base_info():
    return {"channel_version": "1.0.3"}

def send_message(text, max_retries=3):
    """Send text to WeChat, handling session timeout with retries."""
    for attempt in range(max_retries):
        client_id = str(uuid.uuid4())
        ct = get_context_token()
        payload = {
            "msg": {
                "from_user_id": "",
                "to_user_id": TO_USER,
                "client_id": client_id,
                "message_type": "bot",
                "message_state": "finish",
                "item_list": [{"type": "text", "text_item": {"text": text}}],
                "context_token": ct,
            },
            "base_info": build_base_info(),
        }
        body = json.dumps(payload)
        req = urllib.request.Request(
            f"{BASE_URL}/ilink/bot/sendmessage",
            data=body.encode(),
            headers={
                "Content-Type": "application/json",
                "Authorization": f"Bearer {TOKEN}",
                "Token": TOKEN,
            },
            method="POST",
        )
        try:
            with urllib.request.urlopen(req, timeout=15) as r:
                resp = json.loads(r.read())
            errcode = resp.get("errcode", 0)
            if errcode == 0:
                return resp, 200
            elif errcode == -14:
                # Session timeout - wait and retry
                print(f"  [attempt {attempt+1}] session timeout, waiting 5s...")
                time.sleep(5)
                continue
            else:
                return resp, 200  # non-zero but not timeout
        except urllib.error.HTTPError as e:
            body_err = e.read().decode()[:200]
            return {"error": body_err}, e.code
        except Exception as e:
            return {"error": str(e)}, -1
    return {"error": "max retries exceeded"}, -1

    patterns = []
