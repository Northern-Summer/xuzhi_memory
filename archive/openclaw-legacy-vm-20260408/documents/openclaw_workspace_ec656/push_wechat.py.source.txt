import json
import sys
import urllib.request
import urllib.error
from pathlib import Path

GATEWAY = "http://127.0.0.1:18789"
SESSION_KEY = "agent:main:openclaw-weixin:direct:o9cq80z9eorqjasg6hb1w-cc4-po@im.wechat"

def rpc(method, params=None):
    payload = {"method": method, "params": params or {}, "jsonrpc": "2.0", "id": 1}
    req = urllib.request.Request(
        f"{GATEWAY}/rpc",
        data=json.dumps(payload).encode(),
        headers={"Content-Type": "application/json"},
    )
    try:
        with urllib.request.urlopen(req, timeout=10) as r:
            return json.loads(r.read())
    except Exception as e:
        print(f"RPC failed: {e}", file=sys.stderr)
        return None

def send_message(content):
    # Send via sessions_send to WeChat session
    result = rpc("sessions.send", {
        "sessionKey": SESSION_KEY,
        "message": content,
        "timeoutSeconds": 30,
    })
    return result

if __name__ == "__main__":
    if len(sys.argv) < 2:
        # Default: generate expert tracker summary
        tracker_path = Path.home() / ".xuzhi_memory" / "expert_tracker" / "activity.json"
        if tracker_path.exists():
            with open(tracker_path) as f:
                data = json.load(f)
            activities = data.get("activities", {})
            
            # Find recent updates (today)
            recent = []
            for key, info in activities.items():
                updated = info.get("updated_at", "")
                if "2026-03-27" in updated:
                    recent.append(info)
            
            if recent:
                lines = ["【Ξ 前沿追踪】", ""]
                for item in recent[:3]:
                    dept = item.get("dept", "")
                    expert = item.get("expert", "")
                    title = item.get("latest_title", "")
                    url = item.get("latest_url", "")
                    lines.append(f"• [{dept}] {expert}: {title}")
                    if url:
                        lines.append(f"  {url}")
                content = "\n".join(lines)
            else:
                content = "【Ξ 前沿追踪】\n今日暂无新的专家活动记录。"
        else:
            content = "【Ξ 前沿追踪】\nexpert_tracker数据不可用。"
    else:
        content = sys.argv[1]

    result = send_message(content)
    if result:
        print(f"Message sent: {result}")
    else:
        print("Send failed", file=sys.stderr)
        sys.exit(1)
