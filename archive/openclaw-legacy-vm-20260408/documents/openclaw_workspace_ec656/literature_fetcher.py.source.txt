# literature_fetcher.py — 离线文献采集器
"""
策略：
1. 使用 Google Scholar 镜像搜索（无需 API key）
2. 使用 Semantic Scholar 批量搜索（减少请求）
3. 离线存储，避免重复请求

核心文献优先：
- McCarthy & Hayes (1969) — 框架问题原始文献
- Pylyshyn (1987) — 框架问题经典表述
- Brandom (1994) — 规范语义学
"""
import json
import urllib.request
import urllib.parse
import time
from pathlib import Path
from datetime import datetime, timezone

HOME = Path.home()
RESEARCH_DIR = HOME / ".xuzhi_memory" / "research_v2"
LITERATURE_DIR = RESEARCH_DIR / "literature"
CACHE_FILE = RESEARCH_DIR / "literature_cache.json"


def search_semantic_scholar_batch(queries: list[str], delay: float = 2.0) -> list[dict]:
    """批量搜索 Semantic Scholar（带延迟）"""
    all_results = []
    
    for i, query in enumerate(queries):
        print(f"[{i+1}/{len(queries)}] 搜索: {query[:50]}...")
        
        url = "https://api.semanticscholar.org/graph/v1/paper/search"
        params = {
            "query": query,
            "limit": 10,
            "fields": "title,authors,year,abstract,citationCount,venue,openAccessPdf,url"
        }
        
        full_url = f"{url}?{urllib.parse.urlencode(params)}"
        
        try:
            with urllib.request.urlopen(full_url, timeout=30) as response:
                data = json.loads(response.read().decode())
                results = data.get("data", [])
                for r in results:
                    r["search_query"] = query
                all_results.extend(results)
                print(f"  ✓ 找到 {len(results)} 篇")
        except Exception as e:
            print(f"  ✗ 失败: {e}")
        
        time.sleep(delay)
    
    return all_results


def deduplicate(papers: list[dict]) -> list[dict]:
    """按标题去重"""
    seen = set()
    unique = []
    for p in papers:
        title_key = p.get("title", "").lower().strip()[:80]
        if title_key not in seen and title_key:
            seen.add(title_key)
            unique.append(p)
    return unique


def calculate_scores(papers: list[dict], keywords: list[str]) -> list[dict]:
    """计算相关性和质量评分"""
    for p in papers:
        title = p.get("title", "").lower()
        abstract = p.get("abstract", "").lower() or ""
        text = title + " " + abstract
        
        # 相关性评分
        relevance = 0.0
        for kw in keywords:
            if kw.lower() in title:
                relevance += 0.3
            if kw.lower() in abstract:
                relevance += 0.1
        p["relevance_score"] = min(relevance, 1.0)
        
        # 质量评分
        quality = 0.0
        citations = p.get("citationCount", 0)
        if citations > 0:
            import math
            quality += min(math.log10(citations + 1) / 4, 0.4)
        
        year = p.get("year", 2000)
        if year >= 2023:
            quality += 0.2
        elif year >= 2020:
            quality += 0.15
        
        if p.get("openAccessPdf"):
            quality += 0.1
        
        p["quality_score"] = min(quality, 1.0)
        p["combined_score"] = p["relevance_score"] * 0.6 + p["quality_score"] * 0.4
    
    return papers


def save_results(papers: list[dict], filename: str = "search_results.json"):
    """保存结果"""
    path = RESEARCH_DIR / filename
    with open(path, "w") as f:
        json.dump({
            "timestamp": datetime.now(timezone.utc).isoformat(),
            "count": len(papers),
            "papers": papers
        }, f, indent=2, ensure_ascii=False)
    print(f"✓ 已保存 {len(papers)} 篇文献到 {path}")


# 核心搜索查询
CORE_QUERIES = [
    # 框架问题
    "McCarthy Hayes frame problem 1969",
    "frame problem artificial intelligence Pylyshyn",
    "knowledge representation frame problem",
    
    # 规范语义学
    "Brandom inferentialism normative",
    "normative semantics reasoning",
    
    # 多智能体 + 认知
    "multi-agent system failure epistemology",
    "granularity representation abstraction AI",
    "agent architecture knowledge level Newell",
]

KEYWORDS = [
    "frame problem", "knowledge representation", "granularity",
    "multi-agent", "epistemology", "normative", "inferentialism",
    "reasoning", "abstraction", "cognitive architecture"
]


if __name__ == "__main__":
    print("文献采集器启动")
    print("=" * 50)
    
    # 搜索
    papers = search_semantic_scholar_batch(CORE_QUERIES, delay=3.0)
    
    # 去重
    papers = deduplicate(papers)
    print(f"\n去重后: {len(papers)} 篇")
    
    # 评分
    papers = calculate_scores(papers, KEYWORDS)
    
    # 排序
    papers.sort(key=lambda x: x["combined_score"], reverse=True)
    
    # 保存
    save_results(papers[:50])
    
    # 显示 top 10
    print("\n" + "=" * 50)
    print("Top 10:")
    for i, p in enumerate(papers[:10], 1):
        print(f"\n{i}. {p.get('title', 'N/A')[:60]}...")
        print(f"   年份: {p.get('year')} | 引用: {p.get('citationCount', 0)}")
        print(f"   相关性: {p.get('relevance_score', 0):.2f} | 质量: {p.get('quality_score', 0):.2f}")
