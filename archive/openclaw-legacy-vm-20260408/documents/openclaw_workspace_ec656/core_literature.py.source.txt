class CoreLiteratureSearcher:
    """核心文献定位器"""
    
    def __init__(self):
        self.found_papers = []
        self.citation_network = {}
    
    def search_paper(self, paper_info: dict) -> Optional[dict]:
        """搜索单篇核心文献"""
        query = paper_info["search_query"]
        
        # 使用 Semantic Scholar API
        url = "https://api.semanticscholar.org/graph/v1/paper/search"
        params = {
            "query": query,
            "limit": 5,
            "fields": "title,authors,year,abstract,citationCount,venue,openAccessPdf,url,externalIds"
        }
        
        full_url = f"{url}?{urllib.parse.urlencode(params)}"
        
        try:
            with urllib.request.urlopen(full_url, timeout=30) as response:
                data = json.loads(response.read().decode())
                results = data.get("data", [])
                
                if results:
                    # 找最匹配的
                    best_match = self._find_best_match(results, paper_info)
                    if best_match:
                        best_match["priority"] = paper_info["priority"]
                        best_match["relevance_note"] = paper_info["relevance"]
                        best_match["core_id"] = paper_info["id"]
                        return best_match
        except Exception as e:
            print(f"搜索失败: {paper_info['id']} - {e}")
        
        return None
    
    def _find_best_match(self, results: list[dict], target: dict) -> Optional[dict]:
        """找到最匹配的结果"""
        target_year = target["year"]
        target_authors = [a.lower() for a in target["authors"]]
        
        for r in results:
            # 检查年份
            if r.get("year") != target_year:
                continue
            
            # 检查作者
            paper_authors = [a.get("name", "").lower() for a in r.get("authors", [])]
            author_match = any(
                any(ta in pa or pa in ta for pa in paper_authors)
                for ta in target_authors
            )
            
            if author_match:
                return r
        
        # 如果精确匹配失败，返回第一个结果
        return results[0] if results else None
    
    def get_citations(self, paper_id: str, limit: int = 20) -> list[dict]:
        """获取引用这篇论文的文献（前向引用）"""
        url = f"https://api.semanticscholar.org/graph/v1/paper/{paper_id}/citations"
        params = {
            "limit": limit,
            "fields": "title,authors,year,abstract,citationCount,venue"
        }
        
        full_url = f"{url}?{urllib.parse.urlencode(params)}"
        
        try:
            with urllib.request.urlopen(full_url, timeout=30) as response:
                data = json.loads(response.read().decode())
                return data.get("data", [])
        except Exception as e:
            print(f"获取引用失败: {e}")
            return []
    
    def get_related_papers(self, paper_id: str, limit: int = 20) -> list[dict]:
        """获取相关论文"""
        url = f"https://api.semanticscholar.org/recommendations/v1/papers/forpaper/{paper_id}"
        params = {
            "limit": limit,
            "fields": "title,authors,year,abstract,citationCount,venue"
        }
        
        full_url = f"{url}?{urllib.parse.urlencode(params)}"
        
        try:
            with urllib.request.urlopen(full_url, timeout=30) as response:
                data = json.loads(response.read().decode())
                return data.get("recommendedPapers", [])
        except Exception as e:
            print(f"获取相关论文失败: {e}")
            return []
    
    def search_all_core(self) -> dict:
        """搜索所有核心文献"""
        results = {
            "found": [],
            "not_found": [],
            "citation_network": {}
        }
        
        for paper_info in CORE_LITERATURE:
            print(f"搜索: {paper_info['id']}")
            
            found = self.search_paper(paper_info)
            if found:
                results["found"].append(found)
                print(f"  ✓ 找到: {found.get('title', 'N/A')}")
                
                # 获取引用网络
                if found.get("paperId"):
                    time.sleep(0.5)  # rate limit
                    citations = self.get_citations(found["paperId"], limit=10)
                    results["citation_network"][paper_info["id"]] = {
                        "citations": citations[:10],
                        "citation_count": len(citations)
                    }
            else:
                results["not_found"].append(paper_info)
                print(f"  ✗ 未找到")
            
            time.sleep(1)  # rate limit
        
        # 保存结果
        with open(CORE_LIT_FILE, "w") as f:
            json.dump(results, f, indent=2, ensure_ascii=False)
        
        return results


def time_sleep(seconds):
    """避免在 module 层面 import time"""
    import time
    time.sleep(seconds)
