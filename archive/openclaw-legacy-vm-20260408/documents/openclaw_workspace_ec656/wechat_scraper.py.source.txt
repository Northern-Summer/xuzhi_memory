# 微信公众号文章搜索 + 抓取工具
# 使用 curl_cffi 模拟 Chrome TLS 指纹，反反爬
# 用法: python3 wechat_scraper.py "<关键词>" [最大结果数]

from curl_cffi import requests
import re
import sys
import json
import urllib.parse
from datetime import datetime

class WeChatScraper:
    """搜狗微信搜索 + 文章抓取"""

    BASE_URL = "https://weixin.sogou.com/weixin"

    def __init__(self):
        self.session = requests.Session(impersonate="chrome")
        self.session.headers.update({
            "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/120.0.0.0 Safari/537.36",
            "Accept": "text/html,application/xhtml+xml,application/xml;q=0.9,*/*;q=0.8",
            "Accept-Language": "zh-CN,zh;q=0.9,en;q=0.8",
            "Referer": "https://weixin.sogou.com/",
        })

    def search(self, query, max_results=10):
        """搜索公众号文章，返回文章列表"""
        url = f"{self.BASE_URL}?type=2&query={urllib.parse.quote(query)}&ie=utf8"
        r = self.session.get(url, timeout=15)
        r.raise_for_status()
        return self._parse_results(r.text, query)

    def _parse_results(self, html, query):
        """从搜狗微信搜索页解析文章列表"""
        articles = []
        seen_titles = set()

        # 以 <img> 标签分割，每个区块包含一篇文章
        blocks = re.split(r'<img[^>]+>', html)

        current = {}
        for block in blocks:
            lines = block.strip().split('\n')
            for line in lines:
                # 清洗HTML
                clean = re.sub(r'<[^>]+>', '', line).strip()
                # 过滤JS/无效内容
                if (len(clean) < 5 or len(clean) > 150 or
                    any(k in clean for k in ['var ', 'function', 'document', 'sogou.com',
                                             'http', 'window', 'return', 'if(', 'for(',
                                             'cookie', 'setTimeout', 'setInterval', '_hmt', '_ga'])):
                    continue

                # 提取时间戳（向上下文找）
                ts_match = re.search(r"timeConvert\('(\d+)'\)", html)
                if ts_match:
                    ts = int(ts_match.group(1))
                    current['date'] = datetime.fromtimestamp(ts).strftime('%Y-%m-%d')

                # 提取链接
                link_match = re.search(r'/link\?url=([^"&]+)', block)
                if link_match:
                    current['url'] = f"https://weixin.sogou.com{block[link_match.start():link_match.end()]}"

                # 提取账号名
                account_match = re.search(r'class="account"[^>]*>([^<]+)<', block)
                if account_match:
                    current['account'] = account_match.group(1).strip()

                # 判断是否是标题（含AI关键词）
                if any(k in clean for k in ['Agent', 'AI', '智能体', '多智能', 'LLM', '大模型', '系统']):
                    if clean not in seen_titles and len(clean) > 8:
                        seen_titles.add(clean)
                        if current.get('url'):
                            articles.append({
                                'title': clean,
                                'account': current.get('account', '未知'),
                                'date': current.get('date', None),
                                'url': current['url'],
                                'summary': None
                            })
                        current = {}

        return articles

    def search_by_account(self, account_name, max_results=5):
        """搜索指定公众号的文章"""
        # 先搜公众号
        url = f"{self.BASE_URL}?type=1&query={urllib.parse.quote(account_name)}&ie=utf8"
        r = self.session.get(url, timeout=15)
        r.raise_for_status()
        # type=1 搜公众号，type=2 搜文章
        # 简单处理：用关键词拼接
        return self.search(account_name, max_results)

    def get_article_content(self, url, timeout=15):
        """
        通过搜狗重定向获取文章正文
        注意：部分文章需要微信Cookie才能看全文
        """
        try:
            # 先访问搜狗重定向页
            r = self.session.get(url, timeout=timeout, allow_redirects=True)
            
            # 提取最终URL（如果是微信域名，可能直接拿到内容）
            final_url = r.url
            
            if 'mp.weixin.qq.com' in final_url:
                # 通过搜狗代理访问
                text = r.text
            else:
                text = r.text
            
            # 尝试提取正文
            patterns = [
                r'id="js_content"[^>]*>(.*?)id="js_pc_qr_code"',
                r'<div class="rich_media_content[^"]*"[^>]*>(.*?)</div>',
            ]
            
            for pat in patterns:
                m = re.search(pat, text, re.DOTALL)
                if m:
                    raw = m.group(1)
                    text = re.sub(r'<[^>]+>', '', raw)
                    text = re.sub(r'\s+', ' ', text).strip()
                    return text[:5000]  # 限制长度
            
            # 备选：提取所有段落
            paras = re.findall(r'<p[^>]*>([^<]+)<', text)
            content = ' '.join(p.strip() for p in paras if len(p.strip()) > 20)
            return content[:5000] if content else None
            
        except Exception as e:
            return None


def main():
    if len(sys.argv) < 2:
        print("用法: python3 wechat_scraper.py <关键词> [最大结果数]")
        print("示例: python3 wechat_scraper.py '多智能体 AI Agent' 8")
        sys.exit(1)

    query = sys.argv[1]
    max_results = int(sys.argv[2]) if len(sys.argv) > 2 else 8

    scraper = WeChatScraper()
    print(f"🔍 搜索微信公众号文章: {query}\n")

    articles = scraper.search(query, max_results)

    if not articles:
        print("未找到结果")
        sys.exit(0)

    print(f"📰 找到 {len(articles)} 篇文章:\n")
    print(f"{'序号':<4} {'日期':<12} {'来源':<15} {'标题'}")
    print("-" * 80)
    
    for i, a in enumerate(articles, 1):
        account = (a.get('account') or '未知')[:12]
        date = (a.get('date') or '未知日期')[:10]
        title = a['title'][:50]
        print(f"{i:<4} {date:<12} @{account:<12} {title}")
    
    print(f"\n✅ 完整结果已保存，可直接使用链接访问原文")
    return articles


if __name__ == "__main__":
    articles = main()
    
    # 如果是JSON模式（传入 --json 参数）
    if len(sys.argv) > 1 and '--json' in sys.argv:
        print(json.dumps(articles, ensure_ascii=False, indent=2))
