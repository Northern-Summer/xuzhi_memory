#!/usr/bin/env python3
"""
ETP GitHub Discussions Monitor
Track discussions in teorth/equational_theories
"""

import urllib.request
import json
import os
from datetime import datetime
from pathlib import Path

# GitHub API (public, no auth needed for read-only)
REPO = "teorth/equational_theories"
API_BASE = f"https://api.github.com/repos/{REPO}"

DATA_DIR = Path.home() / ".xuzhi_memory" / "agents" / "delta" / "community_observations"

def get_today_file():
    today = datetime.now().strftime("%Y-%m-%d")
    return DATA_DIR / f"github_etp_{today}.jsonl"

def github_request(endpoint):
    """Make GitHub API request"""
    url = f"{API_BASE}/{endpoint}"
    headers = {
        "Accept": "application/vnd.github.v3+json",
        "User-Agent": "Delta-CommunityMonitor"
    }
    
    # Add token if available (higher rate limit)
    token = os.environ.get('GITHUB_TOKEN')
    if token:
        headers["Authorization"] = f"token {token}"
    
    req = urllib.request.Request(url, headers=headers)
    
    with urllib.request.urlopen(req) as response:
        return json.loads(response.read().decode())

def get_recent_discussions():
    """Get recent discussions from ETP repo"""
    # Note: GitHub Discussions API requires GraphQL or special preview header
    # For now, fetch recent issues as proxy
    try:
        issues = github_request("issues?state=all&per_page=30")
        
        print(f"  📊 ETP Issues/Discussions: {len(issues)} items")
        
        for item in issues:
            entry = {
                "timestamp": item.get('created_at'),
                "source": "github_etp",
                "type": "issue" if 'pull_request' not in item else "pr",
                "number": item.get('number'),
                "title": item.get('title'),
                "author": item.get('user', {}).get('login'),
                "state": item.get('state'),
                "labels": [l['name'] for l in item.get('labels', [])],
                "body": (item.get('body') or '')[:1000],
                "comments_count": item.get('comments'),
                "url": item.get('html_url'),
                "collected_at": datetime.now().isoformat()
            }
            
            with open(get_today_file(), "a", encoding="utf-8") as f:
                f.write(json.dumps(entry, ensure_ascii=False) + "\n")
        
        return len(issues)
    except Exception as e:
        print(f"  ❌ Error: {e}")
        return 0

if __name__ == "__main__":
    print("🔍 Monitoring ETP GitHub...")
    count = get_recent_discussions()
    print(f"✅ Saved {count} items to {get_today_file()}")
