#!/usr/bin/env python3
"""
Weekly Community Observation Summary
Delta's analysis of collected community data
"""

import json
from pathlib import Path
from datetime import datetime, timedelta
from collections import Counter

DATA_DIR = Path.home() / ".xuzhi_memory" / "agents" / "delta" / "community_observations"

def load_recent_observations(days=7):
    """Load observations from last N days"""
    observations = []
    
    for i in range(days):
        date = (datetime.now() - timedelta(days=i)).strftime("%Y-%m-%d")
        
        for prefix in ["zulip", "github_etp", "mathoverflow"]:
            file = DATA_DIR / f"{prefix}_{date}.jsonl"
            if file.exists():
                with open(file) as f:
                    for line in f:
                        try:
                            observations.append(json.loads(line))
                        except:
                            pass
    
    return observations

def generate_summary():
    """Generate weekly summary report"""
    obs = load_recent_observations(7)
    
    if not obs:
        print("No observations found.")
        return
    
    print("=" * 70)
    print(f"📊 Weekly Community Observation Summary")
    print(f"   Period: Last 7 days | Total: {len(obs)} entries")
    print("=" * 70)
    
    # Source breakdown
    sources = Counter(o.get('source') for o in obs)
    print(f"\n📍 Sources:")
    for source, count in sources.most_common():
        print(f"   • {source}: {count}")
    
    # Zulip analysis
    zulip_obs = [o for o in obs if o.get('source') == 'zulip']
    if zulip_obs:
        streams = Counter(o.get('stream') for o in zulip_obs)
        print(f"\n💬 Zulip - Active Streams:")
        for stream, count in streams.most_common(10):
            print(f"   • #{stream}: {count} messages")
        
        # Hot topics
        topics = Counter((o.get('stream'), o.get('topic')) for o in zulip_obs)
        print(f"\n🔥 Hot Topics:")
        for (stream, topic), count in topics.most_common(5):
            print(f"   • #{stream} > {topic}: {count}")
    
    # ETP GitHub analysis
    github_obs = [o for o in obs if o.get('source') == 'github_etp']
    if github_obs:
        print(f"\n🐙 ETP GitHub Activity:")
        open_items = [o for o in github_obs if o.get('state') == 'open']
        print(f"   • Recent items: {len(github_obs)}")
        print(f"   • Currently open: {len(open_items)}")
        
        labels = []
        for o in github_obs:
            labels.extend(o.get('labels', []))
        if labels:
            label_counts = Counter(labels)
            print(f"   • Top labels: {', '.join(f'{l}({c})' for l, c in label_counts.most_common(5))}")
    
    # MathOverflow analysis
    mo_obs = [o for o in obs if o.get('source') == 'mathoverflow']
    if mo_obs:
        print(f"\n📚 MathOverflow Activity:")
        tags = Counter(o.get('tag') for o in mo_obs)
        for tag, count in tags.most_common():
            print(f"   • [{tag}]: {count} questions")
        
        # High engagement questions
        high_score = sorted(mo_obs, key=lambda x: x.get('score', 0), reverse=True)[:3]
        if high_score:
            print(f"\n⭐ High Engagement Questions:")
            for q in high_score:
                print(f"   • {q.get('title')[:70]}... (Score: {q.get('score')})")
    
    # Key insights (manual tagging)
    ai_related = [o for o in obs if any(k in str(o).lower() for k in 
        ['alpha', 'ml', 'ai', 'machine learning', 'neural', 'proof', 'automation'])]
    
    if ai_related:
        print(f"\n🤖 AI/ML Related Discussions: {len(ai_related)} entries")
        print("   (Review these for project alignment)")
    
    print("\n" + "=" * 70)
    print(f"💡 Action: Review observations, update project plan accordingly")
    print("=" * 70)

if __name__ == "__main__":
    generate_summary()
